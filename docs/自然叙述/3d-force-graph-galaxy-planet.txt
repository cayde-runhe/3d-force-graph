=== 核心概念 ===

整个 3D 力导向图 = 宇宙星云，两层节点：

  第一层：6 颗大星球（Layer-0 母节点）
    - 每个代表思维链的一个阶段
    - 体积最大，最亮，位置固定

  第二层：88 颗 AOP 小球（Layer-1~5 子节点）
    - 每个代表一个原子作战操作
    - 围绕对应星球分布，大小按连线层级递减
    - 内部有跨星球、跨层级的复杂思维连线

  连线分级（5 级）：
    1 级（实线粗）：星球→星球主链路（step1→2→3→4→5→6→step1 反馈）
    2 级（实线中）：AOP→同星球下一 AOP（层内顺序执行）
    3 级（实线细）：AOP→同星球前序 AOP（层内回查）
    4 级（虚线中）：AOP→跨星球下一 AOP（跨层依赖）
    5 级（虚线细）：AOP→跨星球反馈 AOP（反馈链路）

  星球1：意图理解 — L1（21 AOP：AOP-001~021）

// 根据层级决定球体大小
function getNodeSize(node) {
  if (node.layer === 0) {
    // 母星球：最大
    return node.size;
  } else if (node.layer <= 3) {
    // AOP L1-L3：中等
    return node.size;
  } else {
    // AOP L4-L6：较小
    return node.size;
  }
}

// 根据连线级别决定样式
function getLinkStyle(link) {
  switch (link.level) {
    case 1: // 母星球主链路：白色粗实线
      return { color: "#FFFFFF", width: 3, opacity: 0.9, dashed: false };
    case 2: // 反馈链路：青色虚线细
      return { color: "#44DDFF", width: 2, opacity: 0.5, dashed: true };
    case 3: // AOP 层内链：同色实线中
      return { color: link.color, width: 2, opacity: 0.7, dashed: false };
    case 4: // AOP 反馈闭环：青色虚线细
      return { color: "#44DDFF", width: 1.5, opacity: 0.4, dashed: true };
    case 5: // 关键依赖：极细虚线
      return { color: link.color, width: 1, opacity: 0.3, dashed: true };
    default:
      return { color: "#FFFFFF", width: 1, opacity: 0.5, dashed: false };
  }
}

// 母星球渲染（发光球体 + 两层光晕）
function createPlanet(node) {
  if (node.layer !== 0) return null;

  const group = new THREE.Group();
  const radius = node.size;
  const color = new THREE.Color(node.color);

  // 核心球体
  const sphere = new THREE.Mesh(
    new THREE.SphereGeometry(radius, 32, 32),
    new THREE.MeshStandardMaterial({
      color, emissive: color, emissiveIntensity: 0.9,
      roughness: 0.2, metalness: 0.1
    })
  );
  group.add(sphere);

  // 内层光晕
  group.add(new THREE.Mesh(
    new THREE.SphereGeometry(radius * 1.6, 32, 32),
    new THREE.MeshBasicMaterial({
      color, transparent: true, opacity: 0.12,
      side: THREE.BackSide, blending: THREE.AdditiveBlending, depthWrite: false
    })
  ));

  // 外层光晕
  group.add(new THREE.Mesh(
    new THREE.SphereGeometry(radius * 2.5, 16, 16),
    new THREE.MeshBasicMaterial({
      color, transparent: true, opacity: 0.06,
      side: THREE.BackSide, blending: THREE.AdditiveBlending, depthWrite: false
    })
  ));

  return group;
}

// AOP 小球渲染（发光球体 + 单层光晕）
function createAopNode(node) {
  if (node.layer === 0) return null;

  const group = new THREE.Group();
  const radius = node.size;
  const color = new THREE.Color(node.color);

  const sphere = new THREE.Mesh(
    new THREE.SphereGeometry(radius, 16, 16),
    new THREE.MeshStandardMaterial({
      color, emissive: color, emissiveIntensity: 0.6,
      roughness: 0.3, metalness: 0.1
    })
  );
  group.add(sphere);

  // 单层光晕
  group.add(new THREE.Mesh(
    new THREE.SphereGeometry(radius * 1.8, 16, 16),
    new THREE.MeshBasicMaterial({
      color, transparent: true, opacity: 0.08,
      side: THREE.BackSide, blending: THREE.AdditiveBlending, depthWrite: false
    })
  ));

  return group;
}

// 连线渲染（分级样式）
function createLink(link) {
  if (!link.source || !link.target) return null;

  const s = new THREE.Vector3(link.source.x || 0, link.source.y || 0, link.source.z || 0);
  const t = new THREE.Vector3(link.target.x || 0, link.target.y || 0, link.target.z || 0);

  const curve = new THREE.LineCurve3(s, t);
  const style = getLinkStyle(link);

  const geo = new THREE.BufferGeometry().setFromPoints(curve.getPoints(10));
  const mat = new THREE.LineBasicMaterial({
    color: style.color,
    transparent: true,
    opacity: style.opacity,
    linewidth: style.width,
    dashed: style.dashed
  });

  return new THREE.Line(geo, mat);
}

// 母星球脉冲动画（最亮的那个最强）
graph.onEngineTick(() => {
  const t = Date.now() * 0.002;
  const nodes = graph.graphData().nodes;

  nodes.forEach((node, i) => {
    if (!node.__threeObj) return;

    const sphere = node.__threeObj.children[0];
    if (!sphere || !sphere.material) return;

    const phase = i * (Math.PI / 3);
    const pulse = node.layer === 0
      ? 0.7 + 0.3 * Math.abs(Math.sin(t * 1.2 + phase))  // 母星球更剧烈
      : 0.5 + 0.2 * Math.abs(Math.sin(t + phase));       // AOP 小球轻微

    sphere.material.emissiveIntensity = pulse;

    // AOP 小球轻微浮动
    if (node.layer > 0) {
      const floatY = Math.sin(t * 0.4 + i) * 0.5;
      node.__threeObj.position.y = (node.y || 0) + floatY;
    }
  });
});

// 计算每个母星球的 AOP 范围
function getAopRange(planetId) {
  switch (planetId) {
    case "母星-意图理解": return { min: 1, max: 21 };
    case "母星-情况判断": return { min: 22, max: 35 };
    case "母星-形成构想": return { min: 36, max: 48 };
    case "母星-制定计划": return { min: 49, max: 66 };
    case "母星-仿真推演": return { min: 67, max: 81 };
    case "母星-效果评估": return { min: 82, max: 88 };
    default: return { min: 0, max: 0 };
  }
}

// 根据母星球过滤连线透明度
function getLinkOpacityByPlanet(link, planetId, aopRange) {
  const sId = link.source.id || link.source;
  const tId = link.target.id || link.target;

  // 主链路（level 1）
  if (link.level === 1) {
    if (sId === planetId || tId === planetId) return 0.95;
    return 0.08;
  }
  // AOP 链（level 3）
  if (link.level === 3) {
    if (sId.startsWith("AOP-") && tId.startsWith("AOP-")) {
      const sNum = parseInt(sId.split("-")[1]);
      const tNum = parseInt(tId.split("-")[1]);
      if (sNum >= aopRange.min && sNum <= aopRange.max &&
          tNum >= aopRange.min && tNum <= aopRange.max) return 0.9;
    }
    return 0.04;
  }
  // 反馈 + 依赖（level 4, 5）
  if (link.level === 4 || link.level === 5) {
    if (sId.startsWith("AOP-") && tId.startsWith("AOP-")) {
      const sNum = parseInt(sId.split("-")[1]);
      const tNum = parseInt(tId.split("-")[1]);
      if ((sNum >= aopRange.min && sNum <= aopRange.max) ||
          (tNum >= aopRange.min && tNum <= aopRange.max)) return 0.5;
    }
    return 0.04;
  }
  return 0.1;
}

// 3D 环绕轨道更新（在 onEngineTick 中调用）

function updateOrbitCamera() {
  if (!orbitState.active) return;

  // 全景模式
  if (currentView === 0) {
    updateOverviewOrbit();
    return;
  }

  // 母星球环绕模式
  const planetId = orbitState.targetNode;
  if (!planetId) return;

  const cfg = getOrbitConfig(planetId);
  const planetPos = getNodePos(planetId);

  orbitState.angle += orbitState.speed;

  const t = orbitState.angle;
  const R = orbitState.radius;
  const tx = orbitState.tiltX;
  const tz = orbitState.tiltZ;

  // 3D 椭圆轨道：以母星球为中心的倾斜轨道
  // x 和 z 形成椭圆，y 有倾角倾斜
  const rawX = Math.sin(t) * R;
  const rawZ = Math.cos(t) * R;
  const rawY = Math.cos(t) * R * 0.3; // 上下起伏，让轨道有立体感

  // 应用轨道倾角
  const camX = rawX + rawY * tx;
  const camY = rawY * Math.cos(tx) + rawZ * Math.sin(tz);
  const camZ = rawZ * Math.cos(tz) - rawY * Math.sin(tz);

  const targetX = planetPos.x;
  const targetY = planetPos.y;
  const targetZ = planetPos.z;

  graph.cameraPosition(
    { x: targetX + camX, y: targetY + camY, z: targetZ + camZ },
    { x: targetX, y: targetY, z: targetZ },
    0
  );
}

// 视图面板默认隐藏，进入视图1~6 后显示
document.getElementById("view-panel").style.display = "none";

const galaxyGraph = new ForceGraph3D(document.getElementById("3d-graph"))

  .graphData(steps88Data)

  .nodeThreeObject(node => node.layer === 0 ? createPlanet(node) : createAopNode(node))
  .nodeThreeObjectExtend(false)
  .nodeVal(node => node.layer === 0 ? 18 : node.size)
  .nodeColor(node => node.color)
  .nodeLabel(node => node.layer === 0
    ? `${node.label}\n${node.aopRange}`
    : `AOP-${String(node.aop).padStart(3, '0')} ${node.label}`)

  .linkColor(link => getLinkStyle(link).color)
  .linkWidth(link => getLinkStyle(link).width)
  .linkOpacity(link => getLinkStyle(link).opacity)
  .linkLineDashed(link => getLinkStyle(link).dashed ? [4, 3] : null)

  .nodePosition(node => ({ x: node.x, y: node.y, z: node.z }))

  .cameraPosition({ x: 200, y: 80, z: 180 }, { x: 0, y: 0, z: 0 }, 0)
  .backgroundColor("#050510")

  // 保留力导向（弱排斥力，防止节点重叠）
  .d3Force("charge", d3.forceManyBody().strength(-30))
  .d3Force("collide", d3.forceCollide().radius(node => node.layer === 0 ? 25 : 8))

  .onNodeClick(handleNodeClick)
  .onNodeHover(handleNodeHover)
  .onBackgroundClick(handleBackgroundClick)
  .onEngineTick(updatePulse)

  .warmupTicks(100)
  .cooldownTicks(Infinity);
